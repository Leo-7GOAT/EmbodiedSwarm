cmake_minimum_required(VERSION 3.16)
project(embodied_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_SANITIZERS "Address/undefined behavior sanitizers" OFF)
find_package(Eigen3 REQUIRED)
add_library(embodied_core src/core.cpp)
target_include_directories(embodied_core PUBLIC include)
target_link_libraries(embodied_core PUBLIC Eigen3::Eigen)
target_compile_options(embodied_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ENABLE_SANITIZERS)
  target_compile_options(embodied_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(embodied_core PUBLIC -fsanitize=address,undefined)
endif()
find_package(GTest QUIET)
if(NOT GTest_FOUND)
  add_subdirectory(/usr/src/googletest ${CMAKE_BINARY_DIR}/googletest EXCLUDE_FROM_ALL)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE embodied_core GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(core_tests)
add_executable(core_cli src/cli.cpp)
target_link_libraries(core_cli PRIVATE embodied_core)
